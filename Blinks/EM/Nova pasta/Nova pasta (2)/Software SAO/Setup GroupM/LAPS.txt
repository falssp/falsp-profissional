<#
.SYNOPSIS
  Deploying LAPS on GroupM devices

.DESCRIPTION
  This script will install LAPS on device (if not installed) and/or create/modify local user account designated for LAPS

.PARAMETER
    No PARAMETER

.INPUTS
  No INPUTS

.OUTPUTS
  Script will log process to $var_LogFile (SYSTEM Temp folder with name [COMPUTERNAME]_LAPS.log)
  If MSI install is triggered this will log MSI process to $var_LogFileMSI (SYSTEM Temp folder with name [COMPUTERNAME]_LAPS_MSI.log)

.NOTES
  Version:        1.0
  Author:         Bas Swildens / Bas.Swildens@GroupM.com / Bas.Swildens@kyndryl.com
  Creation Date:  20220514
#>
#requires -version 5
$ErrorActionPreference = 'SilentlyContinue'

function Write-Log {
   Param ([string]$logstring)
   Write-Output "$(Get-Date -Format 'yyyyMMdd-HHmmss') - $logstring"
   Add-content $var_LogFile -value "$(Get-Date -Format 'yyyyMMdd-HHmmss') - $logstring"
}

Function Get-InstalledSoftware {
$var_ArrInstalledSoftware = @()
foreach($obj in (Get-ChildItem "HKLM:\Software\Microsoft\Windows\CurrentVersion\Uninstall")) {
$HashTab = $NULL
$HashTab = [ordered]@{
	Displayname = $obj.GetValue('DisplayName')
	Version = $obj.GetValue('DisplayVersion')
	}
$var_ArrInstalledSoftware += New-Object PSObject -Property $HashTab
}
return $var_ArrInstalledSoftware
}

#Defaults
$DefaultVariables=Get-Variable
$var_DNSRoot = $(Get-WmiObject -Namespace root\cimv2 -Class Win32_ComputerSystem).Domain
$var_TEMPfolder = $([System.Environment]::GetEnvironmentVariable('TEMP','Machine'))
$var_LogFile = "$($var_TEMPfolder)\${env:COMPUTERNAME}_LAPS.log"
$var_LogFileMSI = "$($var_TEMPfolder)\${env:COMPUTERNAME}_LAPS_MSI.log"
$var_Installer = "\\$($var_DNSRoot)\SYSVOL\$($var_DNSRoot)\LAPS\LAPS.x64.msi"
$var_User = "ManagedAdmin"
$var_UserLocal = "$env:COMPUTERNAME\$var_User"
$var_UserDescription = "This local admin account will be managed by LAPS"
$var_UserInfo = Get-LocalUser $var_User
$var_LocalAdministratorsGroupName = (Get-WmiObject win32_group -Filter "Domain='$env:computername' and SID='S-1-5-32-544'").Name
$var_LocalAdministratorsGroupMembers = Get-LocalGroupMember -Name $var_LocalAdministratorsGroupName | Select-Object -ExpandProperty name
$var_LocalOrigAdminSID = (Get-WmiObject Win32_Account -Filter "LocalAccount=True AND SID LIKE '%500%'").SID
$var_UserPassword = ConvertTo-securestring $var_LocalOrigAdminSID -asplaintext -force
$var_ApplicationName = "Local Administrator Password Solution"
$var_ApplicationVersion = "6.2.0.0"

if (Test-Path -LiteralPath "$var_LogFile" -PathType Leaf) { Move-Item -Path "$var_LogFile" -Destination "$var_LogFile.bak" -Force }

Write-Log "Start script $($MyInvocation.MyCommand.Path)"

Write-Log "Check if [$var_UserLocal] health (existence, enabled, description, administrator)"
IF ($($var_UserInfo)) {
	Write-Log "[$var_UserLocal] exists"
	IF ($var_LocalAdministratorsGroupMembers -Contains "$var_UserLocal") {
		Write-Log "[$var_UserLocal] is a local administrator"
	} Else {
		Write-Log "Adding [$var_UserLocal] to local group [$var_LocalAdministratorsGroupName]"
		Add-LocalGroupMember -Group ($var_LocalAdministratorsGroupName) -Member "$var_User"
	}
	IF ($($var_UserInfo).Enabled) {
		Write-Log "[$var_UserLocal]'s account is enabled"
		} Else {
		Write-Log "[$var_UserLocal]'s account is disabled: enable user"
		Enable-LocalUser -Name "$var_User"
		}
	IF ($($var_UserInfo).Description -eq $var_UserDescription) {
		Write-Log "[$var_UserLocal]'s Description is default"
		} Else {
		Write-Log "[$var_UserLocal]'s Description is not default: modifying"
		Set-LocalUser -Name "$var_User" -Description "$var_UserDescription"
		}
	} Else {
		Write-Log "User does not exist: create local user [$var_UserLocal]"
		New-LocalUser "$var_User" -Password $var_UserPassword -FullName "$var_User" -Description "$var_UserDescription" | Set-LocalUser -PasswordNeverExpires $false
		Add-LocalGroupMember -Group ($var_LocalAdministratorsGroupName) -Member "$var_User"
}

Write-Log "Check if [$var_ApplicationName], version [$var_ApplicationVersion] is installed on computer."
$var_Installed = $NULL
$var_Installed =  Get-InstalledSoftware | Where-Object { $_.Displayname -eq $var_ApplicationName -And $_.Version -ge $var_ApplicationVersion }
IF ([string]::IsNullOrWhitespace($var_Installed)) {
 Write-Log "[$var_ApplicationName] is NOT installed. Deploying [$var_Installer]"
 $MSIArguments = "/i ""$var_Installer"" /quiet /L*v ""$var_LogFileMSI"""
 $Process = (Start-Process "msiexec.exe" -ArgumentList "$MSIArguments" -Verb RunAs -Wait -PassThru)
 $ProcessExitCode = $Process.ExitCode
 Write-Log "Deployment [$var_ApplicationName] exited with exit code $ProcessExitCode"
} Else {
 Write-Log "[$var_ApplicationName] is Installed"
}

$Variables_Used = Get-Variable -Exclude ($DefaultVariables.Name + "DefaultVariables")
Foreach ($Variable_Used in $Variables_Used) { Write-Log "Variable name [$($Variable_Used.Name)] Variable value [$($Variable_Used.Value)]" }

Write-Log "Ending script"
